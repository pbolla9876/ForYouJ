  <script>

    /* ==========================================================================

       5. CANVAS & ANIMATION SYSTEM

       ========================================================================== */

    const canvas = document.getElementById('canvas');

    const ctx = canvas.getContext('2d');


    let width = (canvas.width = window.innerWidth);

    let height = (canvas.height = window.innerHeight);


    window.addEventListener('resize', () => {

      width = canvas.width = window.innerWidth;

      height = canvas.height = window.innerHeight;

    });


    const STAR_COUNT = 200;

    let particles = [];

    let phase = 1;


    class Particle {

      constructor() {

        this.reset();

      }


      reset() {

        this.x = Math.random() * width;

        this.y = Math.random() * height;

        this.size = Math.random() * 2 + 1;

        this.alpha = Math.random() * 0.8 + 0.2;

        this.pulseSpeed = Math.random() * 0.02 + 0.005;

        this.dx = (Math.random() - 0.5) * 0.3;

        this.dy = (Math.random() - 0.5) * 0.3;


        this.targetX = width / 2;

        this.targetY = height / 2;

        this.isShard = false;

        this.angle = Math.random() * Math.PI * 2;

        this.rotationSpeed = (Math.random() - 0.5) * 0.1;

      }


      update() {

        if (phase === 1) {

          this.x += this.dx;

          this.y += this.dy;

          this.alpha += Math.sin(Date.now() * this.pulseSpeed) * 0.01;


          if (this.x < 0) this.x = width;

          if (this.x > width) this.x = 0;

          if (this.y < 0) this.y = height;

          if (this.y > height) this.y = 0;


        } else if (phase === 2) {

          const dx = this.targetX - this.x;

          const dy = this.targetY - this.y;

          this.x += dx * 0.08;

          this.y += dy * 0.08;

          this.angle += this.rotationSpeed;

          this.alpha = Math.min(this.alpha + 0.02, 0.9);

        }

      }


      draw() {

        ctx.save();

        ctx.translate(this.x, this.y);


        if (phase === 1 || !this.isShard) {

          ctx.beginPath();

          ctx.arc(0, 0, this.size, 0, Math.PI * 2);

          ctx.fillStyle = `rgba(220, 235, 255, ${Math.max(0, this.alpha)})`;

          ctx.shadowBlur = 8;

          ctx.shadowColor = 'rgba(180, 215, 255, 0.8)';

          ctx.fill();

        } else {

          ctx.rotate(this.angle);

          ctx.beginPath();

          ctx.moveTo(0, -this.size * 3);

          ctx.lineTo(this.size * 2, this.size * 2);

          ctx.lineTo(-this.size * 2, this.size * 1.5);

          ctx.closePath();

          ctx.fillStyle = `rgba(230, 245, 255, ${this.alpha})`;

          ctx.strokeStyle = 'rgba(255, 255, 255, 0.6)';

          ctx.lineWidth = 1;

          ctx.stroke();

          ctx.fill();

        }


        ctx.restore();

      }

    }


    for (let i = 0; i < STAR_COUNT; i++) {

      particles.push(new Particle());

    }


    function render() {

      ctx.clearRect(0, 0, width, height);


      if (phase < 3) {

        particles.forEach((p) => {

          p.update();

          p.draw();

        });

      }


      requestAnimationFrame(render);

    }

    render();


    /* ==========================================================================

       6. MAIN TIMELINE & REVEAL SEQUENCE (GSAP)

       ========================================================================== */

    const tl = gsap.timeline();


    function startSequence() {

      phase = 1;


      tl.to({}, {

        duration: 2.5,

        onComplete: () => {

          phase = 2;

          particles.forEach((p) => {

            p.isShard = true;

            p.size = Math.random() * 4 + 2;

          });

        }

      });


      tl.to({}, {

        duration: 2.5,

        onComplete: () => {

          phase = 3;


          gsap.to('#flash', { opacity: 0.8, duration: 0.1, yoyo: true, repeat: 1 });


          gsap.to('#polaroid', {

            opacity: 1,

            y: 0,

            scale: 1,

            duration: 1.2,

            ease: 'back.out(1.4)'

          });

        }

      });


      tl.to({}, {

        duration: 1.2,

        onComplete: () => {

          document.getElementById('photo').classList.add('developed');

        }

      });


      tl.to('#herName', {

        opacity: 1,

        y: 0,

        duration: 1.2,

        ease: 'power2.out',

        delay: 1.5

      })

      .to('#herCaption', {

        opacity: 1,

        y: 0,

        duration: 1,

        ease: 'power2.out'

      }, '-=0.4')

      .to('#replayBtn', {

        opacity: 1,

        pointerEvents: 'auto',

        duration: 0.8

      })

      .add(() => {

        spawnHeartsLoop();

      });

    }


    function spawnHeartsLoop() {

      setInterval(() => {

        const heart = document.createElement('div');

        heart.classList.add('floating-heart');

        heart.innerHTML = '♥';

        

        const rect = document.getElementById('polaroid').getBoundingClientRect();

        heart.style.left = `${rect.left + Math.random() * rect.width}px`;

        heart.style.top = `${rect.bottom - 20}px`;

        

        document.body.appendChild(heart);


        gsap.to(heart, {

          y: -100 - Math.random() * 50,

          x: (Math.random() - 0.5) * 40,

          opacity: 0,

          duration: 3 + Math.random() * 2,

          ease: 'power1.out',

          onComplete: () => heart.remove()

        });

      }, 1200);

    }


    document.getElementById('replayBtn').addEventListener('click', () => {

      gsap.set('#polaroid', { opacity: 0, y: -80, scale: 0.9 });

      gsap.set('#herName, #herCaption', { opacity: 0, y: 10 });

      gsap.set('#replayBtn', { opacity: 0, pointerEvents: 'none' });

      document.getElementById('photo').classList.remove('developed');


      particles.forEach(p => p.reset());


      tl.restart();

    });


    window.addEventListener('load', () => {

      startSequence();

    });

  </script>
